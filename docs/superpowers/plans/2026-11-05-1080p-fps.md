# 1080p Downscale and FPS Cap Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add a real 1080p downscale option (keeping `orig`) and an FPS cap (30 / 60 / orig) on every surface: pipeline, CLI, presets, desktop GUI, Web UI, OBS dock and remote mode.

**Architecture:** The pipeline already scales by height (`ProcessingOptions.small` + `small_target_height`). This plan moves the edges (CLI, GUI, remote) from the `small_480` boolean to a height (480/720/1080). FPS is a new `ProcessingOptions.max_frame_rate` cap. It is applied with an `fps=` filter **after** `setpts`, because the `setpts` expression counts source frames. The stored preset token `"1080p"` currently means "orig". A one-time migration, keyed on a `presets_version` marker in `settings.json`, rewrites it to `"orig"`.

**Tech Stack:** Python 3, argparse, Tkinter/ttk (GUI), Gradio (Web UI/remote), FFmpeg filter graphs, pytest, black, isort. Plain HTML/JS for the OBS dock.

**Spec:** `docs/superpowers/specs/2026-11-05-1080p-fps-design.md`

def resolution_to_small(resolution: str) -> tuple[bool, Optional[int]]:
    """Map a resolution value to ``(small, target_height)``.

    ``"orig"`` disables scaling and returns no height; ``"1080p"``/``"480p"``
    return their height; ``"720p"`` and any unexpected value fall back to the
    720p small preset, matching how unknown values were always treated.
    """

    if resolution == "orig":
        return False, None
    if resolution == "1080p":
        return True, 1080
    if resolution == "480p":
        return True, 480
    return True, 720


def parse_fps_token(token: object) -> float:
    """Return the frame-rate cap for *token*: ``0.0`` for ``"orig"``.

    Any other value must be a positive finite number; anything else raises
    :class:`ValueError` so callers can reject it (CLI) or skip it (presets).
    """

    text = str(token).strip().lower()
    if text == "orig":
        return 0.0
    value = float(text)
    if not math.isfinite(value) or value <= 0:
        raise ValueError(f"fps must be positive or 'orig', got {token!r}")
    return value


def format_fps_token(max_frame_rate: Optional[float]) -> str:
    """Render a frame-rate cap as a token: ``"orig"`` when unset or non-positive."""

    if not max_frame_rate or max_frame_rate <= 0:
        return "orig"
    return _format_number(max_frame_rate)


def _stored_presets_version(settings: Mapping[str, object]) -> int:
    """Return the ``presets_version`` marker, treating absent/garbage as 1."""

    try:
        return int(settings.get(PRESETS_VERSION_KEY, 1))  # type: ignore[arg-type]
    except (TypeError, ValueError):
        return 1


def _migrate_v1_entry(entry: object) -> object:
    """Rewrite a version-1 preset dict, whose ``"1080p"`` meant "orig"."""

    if isinstance(entry, Mapping) and entry.get("resolution") == "1080p":
        migrated = dict(entry)
        migrated["resolution"] = "orig"
        return migrated
    return entry

```python
def resolve_output_frame_rate(
    source_fps: float, max_frame_rate: Optional[float]
) -> float:
    """Return the frame rate the output is encoded at.

    The cap only lowers the rate: a source at or below *max_frame_rate* (within
    a small tolerance, so 59.94 fps under a 60 cap is kept) is left alone, and an
    unset or non-positive cap means "orig".
    """

    if max_frame_rate and max_frame_rate > 0 and source_fps > max_frame_rate + 1e-3:
        return float(max_frame_rate)
    return source_fps


def _speed_filter_graph(
    scale_part: Optional[str], source_fps: float, output_fps: float, expression: str
) -> str:
    """Build the speed-path filter graph text.

    The ``setpts`` expression indexes **source** frames (``N``) and divides by
    the link's frame rate, so the source-rate ``fps`` must precede it and the
    cap must follow it; lowering the rate first would shift every chunk edge.
    """

    parts: list[str] = []
    if scale_part:
        parts.append(scale_part)
    parts.append(f"fps={source_fps}")
    parts.append(f"setpts={expression.replace(',', '\\,')}")
    if output_fps != source_fps:
        parts.append(f"fps={output_fps}")
    return ",".join(parts)


def _scale_output_frames(frames: int, source_fps: float, output_fps: float) -> int:
    """Convert a source-rate frame count into output frames under an fps cap."""

    if frames <= 0 or source_fps <= 0 or output_fps >= source_fps:
        return frames
    return int(math.ceil(frames * output_fps / source_fps))
```

def resolve_stored_small_height(stored_height: object, legacy_480: object) -> int:
    """Return the persisted small-video height, migrating the old 480p flag.

    return 480 if bool(legacy_480) else 720
